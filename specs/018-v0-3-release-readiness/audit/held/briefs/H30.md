The part-name sanitizing moves into a small `partName` helper shared by `upload` and `savePart`; an invalid part name now gets 400 (generic "bad request") instead of 500.

// uploadOne posts a single-part multipart upload of name=body into dir
// (a data-root path such as "/") and returns the status code.
func uploadOne(t *testing.T, srvURL, dir, name, body string) int {
	t.Helper()
	buf, ct := multipartBody(t, map[string]string{name: body})
	resp, err := testPost(t, srvURL+"/files/upload?path="+dir, ct, buf)
	if err != nil {
		t.Fatalf("post: %v", err)
	}
	defer resp.Body.Close()
	return resp.StatusCode
}

// requireSymlinkTo fails unless path is still a symlink pointing at target.
func requireSymlinkTo(t *testing.T, path, target string) {
	t.Helper()
	fi, err := os.Lstat(path)
	if err != nil {
		t.Fatalf("lstat %s: %v", path, err)
	}
	if fi.Mode()&os.ModeSymlink == 0 {
		t.Fatalf("%s is no longer a symlink (mode %v)", path, fi.Mode())
	}
	got, err := os.Readlink(path)
	if err != nil || got != target {
		t.Fatalf("readlink %s = %q err=%v, want %q", path, got, err, target)
	}
}

// TestUpload_FinalPathStaysInsideDataRoot checks that an upload's full
// destination is confined to the data root: a part whose name is an
// existing entry that resolves outside the root is refused with a 4xx,
// the file outside is unchanged, and the entry is left in place.
func TestUpload_FinalPathStaysInsideDataRoot(t *testing.T) {
	srvURL, root := newServer(t)
	outsideDir, err := filepath.EvalSymlinks(t.TempDir())
	if err != nil {
		t.Fatalf("eval outside dir: %v", err)
	}
	outside := filepath.Join(outsideDir, "outside.txt")
	if err := os.WriteFile(outside, []byte("outside content"), 0o600); err != nil {
		t.Fatalf("write outside file: %v", err)
	}
	link := filepath.Join(root, "linked.txt")
	if err := os.Symlink(outside, link); err != nil {
		t.Skipf("symlink unsupported: %v", err)
	}

	if code := uploadOne(t, srvURL, "/", "linked.txt", "uploaded content"); code/100 != 4 {
		t.Fatalf("status=%d, want 4xx", code)
	}
	got, err := os.ReadFile(outside)
	if err != nil || string(got) != "outside content" {
		t.Fatalf("outside file changed: got %q err=%v", got, err)
	}
	requireSymlinkTo(t, link, outside)
}
