What changes:

1. `agent/internal/rcon/websocket.go`: a failed WebRcon dial error is redacted at the source. `redactDialErr` replaces the path-escaped and raw password in the error text with `[redacted]`, and drops the `*url.Error` (which holds the full request URL) from the chain while keeping its transport cause for `errors.Is/As`. Because the error is clean where it is made, the players, status and actions log sites need no change.
2. `agent/internal/console/console.go`: the `{kind:"err"}` envelope carries a fixed body (`RCON authentication failed`, `RCON is not enabled for this game`, or `RCON command failed; details are in the agent log`) and the error is logged with `slog.Warn("console rcon", ...)`. Same rule as players/actions ("upstream unavailable").
3. `agent/cmd/main.go`: the agent exits 1 at startup when only one of `--tls-cert` / `--tls-key` is set, instead of quietly listening on plain HTTP. The operator always passes both, so deployed agents are unaffected.
4. `agent/specs.md`: the six statements are aligned with what the code enforces (WebRcon uses `IsAllowed`; `/healthz`/`/metrics` sit behind the mTLS handshake; no dotfile rule in `files`; mod downloads refuse private addresses whatever `allowedHosts` says; `--tls-cert` needs `--tls-key`; three confinement helpers, not one).

// redactedPlaceholder replaces a secret in redacted error text.
const redactedPlaceholder = "[redacted]"

// redactedDialError is a dial failure whose text has had the connection
// secret removed. It unwraps to the transport cause only (the error the
// *url.Error wrapped, e.g. a *net.OpError), never to the *url.Error that
// holds the full request URL.
type redactedDialError struct {
	msg   string
	cause error
}

func (e *redactedDialError) Error() string { return e.msg }
func (e *redactedDialError) Unwrap() error { return e.cause }

// redactDialErr returns err with every occurrence of the given secrets
// replaced by a placeholder. Pass the escaped form before the raw one:
// the escaped form is what appears in a URL. Empty secrets are skipped.
// The returned error keeps the transport cause for errors.Is/As, and
// drops any *url.Error from the chain because it carries the full URL.
func redactDialErr(err error, secrets ...string) error {
	if err == nil {
		return nil
	}
	msg := err.Error()
	for _, s := range secrets {
		if s != "" {
			msg = strings.ReplaceAll(msg, s, redactedPlaceholder)
		}
	}
	cause := err
	var uerr *url.Error
	if errors.As(err, &uerr) {
		cause = uerr.Err
	}
	if cause != nil {
		causeMsg := cause.Error()
		for _, s := range secrets {
			if s != "" && strings.Contains(causeMsg, s) {
				// The cause itself carries a secret: keep the redacted
				// text only.
				cause = nil
				break
			}
		}
	}
	return &redactedDialError{msg: msg, cause: cause}
}

package rcon

import (

// redactSentinel is a unique RCON password for these tests. It holds a
// space, a slash and a semicolon so its url.PathEscape form differs from
// the raw form and both can be checked.
const redactSentinel = "gp-sentinel pw/7f3a;c91"

// closedLoopbackPort returns a loopback TCP port that nothing listens on.
func closedLoopbackPort(t *testing.T) int {
	t.Helper()
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatalf("listen: %v", err)
	}
	port := ln.Addr().(*net.TCPAddr).Port
	if err := ln.Close(); err != nil {
		t.Fatalf("close listener: %v", err)
	}
	return port
}

// requireNoSecret fails if text contains the sentinel in raw, path-escaped
// or query-escaped form.
func requireNoSecret(t *testing.T, text string) {
	t.Helper()
	for _, form := range []string{redactSentinel, url.PathEscape(redactSentinel), url.QueryEscape(redactSentinel)} {
		if strings.Contains(text, form) {
			t.Fatalf("error text contains the RCON password (form %q): %s", form, text)
		}
	}
}

// TestWebSocketDialErrorOmitsPassword checks that a failed WebRcon dial
// returns an error whose text never contains the RCON password, in any
// encoding, while still reading as a dial failure.
func TestWebSocketDialErrorOmitsPassword(t *testing.T) {
	client := NewWebSocket("127.0.0.1", closedLoopbackPort(t), func() (string, error) { return redactSentinel, nil })
	client.dialTimeout = 500 * time.Millisecond
	defer client.Close()

	_, err := client.Exec("status")
	if err == nil {
		t.Fatal("Exec to a closed port should fail")
	}
	requireNoSecret(t, err.Error())
	requireNoSecret(t, fmt.Sprintf("%+v", err))
	if !strings.Contains(err.Error(), "dial") {
		t.Errorf("expected the error to read as a dial failure, got: %v", err)
	}
	if errors.Is(err, ErrAuth) {
		t.Errorf("a dial failure must not be classified as ErrAuth: %v", err)
	}
	var uerr *url.Error
	if errors.As(err, &uerr) {
		t.Errorf("the error chain must not expose the *url.Error holding the request URL: %v", uerr)
	}
}

// TestRedactDialErrKeepsTransportCause checks that redaction removes the
// secret from the text and the *url.Error from the chain, and keeps the
// transport cause reachable with errors.Is.
func TestRedactDialErrKeepsTransportCause(t *testing.T) {
	escaped := url.PathEscape(redactSentinel)
	orig := fmt.Errorf("failed to WebSocket dial: failed to send handshake request: %w", &url.Error{
		Op:  "Get",
		URL: "http://127.0.0.1:28016/" + escaped,
		Err: io.ErrUnexpectedEOF,
	})

	got := redactDialErr(orig, escaped, redactSentinel)
	requireNoSecret(t, got.Error())
	if !strings.Contains(got.Error(), redactedPlaceholder) {
		t.Errorf("expected the placeholder in the redacted text, got: %v", got)
	}
	if !errors.Is(got, io.ErrUnexpectedEOF) {
		t.Errorf("the transport cause must stay reachable with errors.Is: %v", got)
	}
	var uerr *url.Error
	if errors.As(got, &uerr) {
		t.Errorf("the *url.Error must not stay in the chain: %v", uerr)
	}
}

// TestRedactDialErrDropsCauseCarryingSecret checks that when the wrapped
// cause itself carries the secret, redaction keeps only the redacted text.
func TestRedactDialErrDropsCauseCarryingSecret(t *testing.T) {
	orig := fmt.Errorf("handshake: %w", errors.New("server said: bad password "+redactSentinel))

	got := redactDialErr(orig, url.PathEscape(redactSentinel), redactSentinel)
	requireNoSecret(t, got.Error())
	for e := errors.Unwrap(got); e != nil; e = errors.Unwrap(e) {
		requireNoSecret(t, e.Error())
	}
}

// TestRedactDialErrIgnoresEmptySecret checks that an empty password
// leaves the error text as it was instead of inserting placeholders.
func TestRedactDialErrIgnoresEmptySecret(t *testing.T) {
	orig := errors.New("dial tcp 127.0.0.1:28016: connect: connection refused")
	if got := redactDialErr(orig, "", ""); got.Error() != orig.Error() {
		t.Errorf("got %q, want %q", got.Error(), orig.Error())
	}
	if redactDialErr(nil, redactSentinel) != nil {
		t.Error("redactDialErr(nil) must return nil")
	}
}

import (

// TestValidateTLSFlags checks that TLS is enabled only with both the
// certificate and its key, and that a single flag is refused.
func TestValidateTLSFlags(t *testing.T) {
	cases := []struct {
		name      string
		cert, key string
		wantErr   bool
	}{
		{name: "neither", cert: "", key: "", wantErr: false},
		{name: "both", cert: "/tls/tls.crt", key: "/tls/tls.key", wantErr: false},
		{name: "cert without key", cert: "/tls/tls.crt", key: "", wantErr: true},
		{name: "key without cert", cert: "", key: "/tls/tls.key", wantErr: true},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			err := validateTLSFlags(tc.cert, tc.key)
			if (err != nil) != tc.wantErr {
				t.Fatalf("validateTLSFlags(%q, %q) = %v, wantErr %v", tc.cert, tc.key, err, tc.wantErr)
			}
		})
	}
}

// agentMainSubprocessEnv marks the child process that runs main() for
// TestAgentRefusesTLSCertWithoutKey.
const agentMainSubprocessEnv = "GAMEPLANE_AGENT_MAIN_SUBPROCESS"

// TestAgentRefusesTLSCertWithoutKey checks that the agent exits non-zero
// at startup when --tls-cert is given without --tls-key, instead of
// serving plain HTTP.
func TestAgentRefusesTLSCertWithoutKey(t *testing.T) {
	if os.Getenv(agentMainSubprocessEnv) == "1" {
		os.Args = []string{"agent", "--tls-cert=/nonexistent/tls.crt", "--addr=127.0.0.1:0"}
		main()
		return
	}

	ctx, cancel := context.WithTimeout(t.Context(), 20*time.Second)
	defer cancel()
	cmd := exec.CommandContext(ctx, os.Args[0], "-test.run=^TestAgentRefusesTLSCertWithoutKey$")
	cmd.Env = append(os.Environ(), agentMainSubprocessEnv+"=1")
	out, err := cmd.CombinedOutput()
	if ctx.Err() != nil {
		t.Fatalf("agent kept running with --tls-cert and no --tls-key; output:\n%s", out)
	}
	var exitErr *exec.ExitError
	if !errors.As(err, &exitErr) || exitErr.ExitCode() == 0 {
		t.Fatalf("expected a non-zero exit, got err=%v; output:\n%s", err, out)
	}
	if !strings.Contains(string(out), "tls flags") {
		t.Fatalf("expected the exit to come from the TLS flag check; output:\n%s", out)
	}
}
