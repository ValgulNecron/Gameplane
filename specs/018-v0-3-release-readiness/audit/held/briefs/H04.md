~~~~markdown
so the client cert and the CA the agents trust stay a matching pair.
~~~~

~~~~bash
#!/usr/bin/env bash
# Render checks for charts/gameplane that need more than one `helm template`
# run with different values.
#
# Agent mTLS Secrets: the chart owns gameplane-agent-ca and
# gameplane-agent-client only under those default names. When
# api.agentMTLS.caSecretRef.name or api.agentMTLS.clientCertRef.name names
# another Secret, that Secret belongs to the user: the render must contain no
# chart-owned Secret for it, and the API and operator must mount the named
# Secret instead.
#
# Needs helm and python3 with PyYAML. Run from the repository root:
#   make check-chart-render
set -euo pipefail

CHART="${CHART:-charts/gameplane}"
RELEASE="gameplane"
NS="gameplane-system"

tmp="$(mktemp -d)"
trap 'rm -rf "$tmp"' EXIT
failures=0

# assert_mtls <rendered.yaml> <ca-name> <client-name>
# Default names must be rendered as chart-owned Secrets; overridden names
# must not be rendered at all, and no Secret of the default name may appear
# for an overridden slot.
assert_mtls() {
  python3 - "$1" "$2" "$3" <<'PY'
import sys
import yaml

path, ca_name, client_name = sys.argv[1:4]
DEFAULT_CA = "gameplane-agent-ca"
DEFAULT_CLIENT = "gameplane-agent-client"

docs = [d for d in yaml.safe_load_all(open(path)) if d]
secrets = {d["metadata"]["name"] for d in docs if d.get("kind") == "Secret"}
errors = []

for slot, name, default in (("CA", ca_name, DEFAULT_CA),
                            ("client", client_name, DEFAULT_CLIENT)):
    if name == default:
        if default not in secrets:
            errors.append(f"default {slot} Secret {default} is not rendered")
    else:
        if default in secrets:
            errors.append(
                f"{slot} name is overridden to {name}, but the chart still "
                f"renders its own {default} Secret")
        if name in secrets:
            errors.append(
                f"{slot} Secret {name} is named by an override, but the chart "
                "renders a Secret of that name")

want = {"agent-ca": ca_name, "agent-client": client_name}
mounted = 0
for d in docs:
    if d.get("kind") != "Deployment":
        continue
    dep = d["metadata"]["name"]
    for vol in d["spec"]["template"]["spec"].get("volumes") or []:
        if vol.get("name") not in want:
            continue
        mounted += 1
        got = (vol.get("secret") or {}).get("secretName")
        if got != want[vol["name"]]:
            errors.append(
                f"{dep} volume {vol['name']} mounts {got}, "
                f"expected {want[vol['name']]}")
    for c in d["spec"]["template"]["spec"].get("containers") or []:
        for arg in c.get("args") or []:
            if arg.startswith("--agent-ca-secret-name=") and \
                    arg != f"--agent-ca-secret-name={ca_name}":
                errors.append(f"{dep} passes {arg}, expected CA {ca_name}")

if mounted < 4:
    errors.append(
        f"expected the API and operator to mount agent-ca and agent-client "
        f"(4 volumes), found {mounted}")

if errors:
    for e in errors:
        print(f"::error::{e}")
    sys.exit(1)
PY
}

# run_case <label> <ca-name> <client-name> [helm args...]
run_case() {
  local label="$1" ca="$2" client="$3"
  shift 3
  local out="$tmp/render.yaml"
  if ! helm template "$RELEASE" "$CHART" -n "$NS" "$@" > "$out"; then
    echo "FAIL: $label: helm template failed"
    failures=$((failures + 1))
    return
  fi
  if assert_mtls "$out" "$ca" "$client"; then
    echo "ok:   $label"
  else
    echo "FAIL: $label"
    failures=$((failures + 1))
  fi
}

run_case "default agent mTLS names render chart-owned Secrets" \
  gameplane-agent-ca gameplane-agent-client

run_case "overridden CA and client names render no chart-owned Secret" \
  my-agent-ca my-agent-client \
  --set api.agentMTLS.caSecretRef.name=my-agent-ca \
  --set api.agentMTLS.clientCertRef.name=my-agent-client

run_case "overridden client name renders only the chart CA Secret" \
  gameplane-agent-ca my-agent-client \
  --set api.agentMTLS.clientCertRef.name=my-agent-client

# A CA-only override needs the user's CA to sign the chart's client cert.
# Offline there is nothing to look up, so the render must stop with a message
# naming the value, not silently sign with a throwaway CA.
label="overridden CA name without an existing CA Secret stops the render"
if msg="$(helm template "$RELEASE" "$CHART" -n "$NS" \
    --set api.agentMTLS.caSecretRef.name=my-agent-ca 2>&1 >/dev/null)"; then
  echo "FAIL: $label: render succeeded"
  failures=$((failures + 1))
elif ! grep -q 'api.agentMTLS.caSecretRef.name' <<<"$msg"; then
  echo "FAIL: $label: unexpected error: $msg"
  failures=$((failures + 1))
else
  echo "ok:   $label"
fi

if [ "$failures" -ne 0 ]; then
  echo "$failures chart render check(s) failed"
  exit 1
fi
echo "all chart render checks passed"
~~~~
