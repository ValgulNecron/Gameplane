// dnsLabel matches a lowercase RFC 1123 label, the form Kubernetes requires
// for Service port names and so the form every frp proxy name takes.
var dnsLabel = regexp.MustCompile(`^[a-z0-9]([-a-z0-9]{0,61}[a-z0-9])?$`)

// hostLabel matches one label of a DNS hostname, in either case.
var hostLabel = regexp.MustCompile(`^[A-Za-z0-9]([-A-Za-z0-9]{0,61}[A-Za-z0-9])?$`)

// validHost reports whether s is an IP address or a DNS hostname: one or
// more labels joined by dots, with an optional trailing dot, at most 253
// characters.
func validHost(s string) bool {
	if net.ParseIP(s) != nil {
		return true
	}
	name := strings.TrimSuffix(s, ".")
	if name == "" || len(name) > 253 {
		return false
	}
	for _, label := range strings.Split(name, ".") {
		if !hostLabel.MatchString(label) {
			return false
		}
	}
	return true
}

// parsePort parses a TCP/UDP port number in the range 1-65535.
func parsePort(s string) (int, error) {
	p, err := strconv.Atoi(strings.TrimSpace(s))
	if err != nil {
		return 0, fmt.Errorf("parse port: %w", err)
	}
	if p < 1 || p > 65535 {
		return 0, fmt.Errorf("port %d out of range 1-65535", p)
	}
	return p, nil
}

// escapeTomlString escapes s for use inside a TOML basic string ("..."):
// the backslash, the double quote, and every control character, none of
// which TOML allows unescaped in a basic string.
func escapeTomlString(s string) string {
	const hexDigits = "0123456789ABCDEF"
	var b strings.Builder
	b.Grow(len(s))
	for _, r := range s {
		switch r {
		case '\\':
			b.WriteString(`\\`)
		case '"':
			b.WriteString(`\"`)
		case '\b':
			b.WriteString(`\b`)
		case '\t':
			b.WriteString(`\t`)
		case '\n':
			b.WriteString(`\n`)
		case '\f':
			b.WriteString(`\f`)
		case '\r':
			b.WriteString(`\r`)
		default:
			if r < 0x20 || r == 0x7f {
				b.WriteString(`\u00`)
				b.WriteByte(hexDigits[r>>4])
				b.WriteByte(hexDigits[r&0xF])
			} else {
				b.WriteRune(r)
			}
		}
	}
	return b.String()
}

// frpcLine matches one `key = value` line of a rendered frpc config: the
// value is a TOML basic string or a decimal integer.
var frpcLine = regexp.MustCompile(`^([A-Za-z.]+) = ("(?:[^"\\\x00-\x1f\x7f]|\\.)*"|[0-9]+)$`)

// frpcEntry is one parsed line of a rendered frpc config.
type frpcEntry struct {
	key   string
	value string // decoded string value, or the integer as written
}

// parseFrpc splits a rendered frpc config into its [[proxies]] headers and
// key/value lines, failing the test on any other line.
func parseFrpc(t *testing.T, content string) []frpcEntry {
	t.Helper()
	var out []frpcEntry
	for _, line := range strings.Split(content, "\n") {
		if line == "" {
			continue
		}
		if line == "[[proxies]]" {
			out = append(out, frpcEntry{key: line})
			continue
		}
		m := frpcLine.FindStringSubmatch(line)
		if m == nil {
			t.Fatalf("line is not a single key/value pair: %q\nconfig:\n%s", line, content)
		}
		v := m[2]
		if strings.HasPrefix(v, `"`) {
			dec, err := strconv.Unquote(v)
			if err != nil {
				t.Fatalf("value %s does not decode: %v", v, err)
			}
			v = dec
		}
		out = append(out, frpcEntry{key: m[1], value: v})
	}
	return out
}
