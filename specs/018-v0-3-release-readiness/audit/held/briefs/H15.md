## 2. What the change does (for orientation)

- The operator records, on `Module.status.verification`, the signature check that admitted the applied bundle: digest, policy kind (`key`/`keyless`), a policy identifier, and time. It is set only when the source's `spec.verify` ran; cleared when the source has no policy; removed when a check fails.
- A Ready Module whose source declares a policy counts as converged only while that record matches the applied digest and the current policy. So adding or changing a source's policy after install makes the operator pull and verify again.
- The API catalog passes `status.verification.digest`/`mode` through as `verifiedDigest`/`verifiedMode` (pure read/map).
- The dashboard shows the solid "verified" chip only when that record matches the applied digest and the source's current policy kind; otherwise a declared policy shows the existing outline "policy" chip. No new visual.

// verifyPolicy describes a source's signature policy for the Module status.
// mode is "key" or "keyless"; policy identifies the policy's settings so a
// change to them is noticed. Both are empty when the source declares no
// policy. The key-before-keyless order mirrors verify.Build.
func verifyPolicy(src *gameplanev1alpha1.ModuleSource) (mode, policy string) {
	v := src.Spec.Verify
	switch {
	case v == nil:
		return "", ""
	case v.Key != nil:
		policy = "key:" + v.Key.Name
		if v.RequireTransparencyLog {
			policy += "+tlog"
		}
		return "key", policy
	case v.Keyless != nil:
		return "keyless", "keyless:" + v.Keyless.Issuer + " " + v.Keyless.Identity
	default:
		return "", ""
	}
}

// verificationCurrent reports whether the status holds what the source's
// policy requires: nothing when there is no policy, otherwise a recorded
// check of the applied digest under that same policy.
func verificationCurrent(st *gameplanev1alpha1.ModuleStatus, policy string) bool {
	if policy == "" {
		return true
	}
	v := st.Verification
	return v != nil && v.Digest == st.AppliedDigest && v.Policy == policy
}

func byCatalogName(entries []gameplanev1alpha1.ModuleEntry, name string) *gameplanev1alpha1.ModuleEntry {
	for i := range entries {
		if entries[i].Name == name {
			return &entries[i]
		}
	}
	return nil
}

// withModuleReconcilerPolicyVerifier wires a verifier that follows the
// source's declared policy, as verify.Build does: a source without
// spec.verify gets verify.Nop, a source with one gets v.
func withModuleReconcilerPolicyVerifier(fake *fakeOCI, v verify.Verifier) setupReconciler {
	return func(mgr manager.Manager) error {
		return (&ModuleReconciler{
			Client:     mgr.GetClient(),
			Scheme:     mgr.GetScheme(),
			NewFetcher: fakeOCIFetcher(fake),
			NewVerifier: func(_ context.Context, src *gameplanev1alpha1.ModuleSource) (verify.Verifier, error) {
				if src.Spec.Verify == nil {
					return verify.Nop{}, nil
				}
				return v, nil
			},
		}).SetupWithManager(mgr)
	}
}

// setSourceVerify replaces a ModuleSource's spec.verify, retrying on conflict.
func setSourceVerify(t *testing.T, name string, v *gameplanev1alpha1.VerifySpec) {
	t.Helper()
	if err := retry.RetryOnConflict(retry.DefaultRetry, func() error {
		var src gameplanev1alpha1.ModuleSource
		if err := k8sClient.Get(context.Background(), types.NamespacedName{Name: name}, &src); err != nil {
			return err
		}
		src.Spec.Verify = v
		return k8sClient.Update(context.Background(), &src)
	}); err != nil {
		t.Fatalf("set source verify: %v", err)
	}
}

	return &gameplanev1alpha1.VerifySpec{Key: &corev1.LocalObjectReference{Name: secret}}

// expectVerificationRecorded waits for a Ready Module whose status records
// a signature check of the applied digest under the given policy.
func expectVerificationRecorded(t *testing.T, name, policy string) {
	t.Helper()
	eventually(t, func() (bool, string) {
		got := getModule(t, name)
		if got.Status.Phase != gameplanev1alpha1.ModulePhaseReady {
			return false, "phase=" + got.Status.Phase
		}
		v := got.Status.Verification
		if v == nil {
			return false, "no verification recorded"
		}
		if v.Digest != got.Status.AppliedDigest {
			return false, fmt.Sprintf("verification.digest=%q appliedDigest=%q", v.Digest, got.Status.AppliedDigest)
		}
		if v.Policy != policy {
			return false, "verification.policy=" + v.Policy
		}
		return true, ""
	})
}
