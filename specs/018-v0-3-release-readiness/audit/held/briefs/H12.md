// awaitRestoredServer drives the Restore to a terminal phase based on the
// newly-provisioned server: Running → Succeeded, Failed → Failed, otherwise
// keep polling while it starts up, until the restore deadline passes.
func (r *RestoreReconciler) awaitRestoredServer(
	ctx context.Context, rs *gameplanev1alpha1.Restore, gs *gameplanev1alpha1.GameServer,
) (ctrl.Result, error) {
	switch gs.Status.Phase {
	case gameplanev1alpha1.GameServerPhaseRunning:
		now := metav1.Now()
		rs.Status.Phase = gameplanev1alpha1.RestorePhaseSucceeded
		if rs.Status.CompletionTime == nil {
			rs.Status.CompletionTime = &now
		}
		rs.Status.Conditions = upsertCondition(rs.Status.Conditions, metav1.Condition{
			Type:               "Completed",
			Status:             metav1.ConditionTrue,
			Reason:             "Succeeded",
			ObservedGeneration: rs.Generation,
		})
		if err := r.Status().Update(ctx, rs); err != nil {
			return ctrl.Result{}, err
		}
		return ctrl.Result{}, nil
	case gameplanev1alpha1.GameServerPhaseFailed:
		return r.fail(ctx, rs, fmt.Sprintf("restored server %q failed to start", gs.Name))
	default:
		if rs.Status.StartTime == nil {
			now := metav1.Now()
			rs.Status.StartTime = &now
			if err := r.Status().Update(ctx, rs); err != nil {
				return ctrl.Result{}, err
			}
		}
		timeout := r.VolumeSnapshotRestoreTimeout
		if timeout <= 0 {
			timeout = defaultVolumeSnapshotRestoreTimeout
		}
		remaining := timeout - time.Since(rs.Status.StartTime.Time)
		if remaining <= 0 {
			phase := string(gs.Status.Phase)
			if phase == "" {
				phase = "none"
			}
			return r.fail(ctx, rs, fmt.Sprintf(
				"restored server %q did not reach Running within %s (phase: %s)",
				gs.Name, timeout, phase))
		}
		return ctrl.Result{RequeueAfter: min(10*time.Second, remaining)}, nil
	}
}

// ownedRefCopy is one Secret or ConfigMap that the original server owns
// and the restored server gets its own copy of.
type ownedRefCopy struct {
	kind string // refKindSecret or refKindConfigMap
	from string
	to   string
}

// refOwnership is what a lookup of a referenced Secret or ConfigMap found.
type refOwnership int

const (
	refMissing refOwnership = iota
	refOwned
	refNotOwned
)

// restoredRefName names the restored server's copy of an object. A name
// that starts with the original server's name keeps its suffix under the
// new server's name ("smp-env" becomes "smp-restored-env"); any other name
// gets the new server's name as a prefix.
func restoredRefName(origServer, newServer, ref string) string {
	if strings.HasPrefix(ref, origServer+"-") {
		return newServer + strings.TrimPrefix(ref, origServer)
	}
	return newServer + "-" + ref
}

// ownershipOfRef reads the named Secret or ConfigMap in gs's namespace and
// reports whether it is missing, owned by gs, or not owned by gs. It uses
// the same ownership rule the GameServer reconcile applies to these
// references.
func (r *RestoreReconciler) ownershipOfRef(
	ctx context.Context, kind, name string, gs *gameplanev1alpha1.GameServer,
) (refOwnership, error) {
	key := types.NamespacedName{Namespace: gs.Namespace, Name: name}
	if kind == refKindSecret {
		var sec corev1.Secret
		err := r.Get(ctx, key, &sec)
		if apierrors.IsNotFound(err) {
			return refMissing, nil
		}
		if err != nil {
			return refMissing, fmt.Errorf("read Secret %q: %w", name, err)
		}
		if isServerOwnedSecret(&sec, gs) {
			return refOwned, nil
		}
		return refNotOwned, nil
	}
	var cm corev1.ConfigMap
	err := r.Get(ctx, key, &cm)
	if apierrors.IsNotFound(err) {
		return refMissing, nil
	}
	if err != nil {
		return refMissing, fmt.Errorf("read ConfigMap %q: %w", name, err)
	}
	if isServerOwnedConfigMap(&cm, gs) {
		return refOwned, nil
	}
	return refNotOwned, nil
}

// planOwnedRefCopies rewrites the Secret and ConfigMap references in spec
// that a server may only make to objects it owns (spec.env valueFrom refs
// and the tunnel credentials Secret) to the names of the restored server's
// copies, and lists the copies to make. A reference to an object that
// doesn't exist is left as it is. A reference to an object the original
// server doesn't own returns a failure message, because the restored
// server may not use it either.
func (r *RestoreReconciler) planOwnedRefCopies(
	ctx context.Context, orig *gameplanev1alpha1.GameServer, newName string,
	spec *gameplanev1alpha1.GameServerSpec,
) ([]ownedRefCopy, string, error) {
	var copies []ownedRefCopy
	renamed := map[string]string{}
	resolve := func(kind, name string) (string, string, error) {
		key := kind + "/" + name
		if to, ok := renamed[key]; ok {
			return to, "", nil
		}
		state, err := r.ownershipOfRef(ctx, kind, name, orig)
		if err != nil {
			return "", "", err
		}
		switch state {
		case refMissing:
			renamed[key] = name
			return name, "", nil
		case refNotOwned:
			return "", fmt.Sprintf(
				"%s %q referenced by server %q is not owned by it; a restored server may only reference objects it owns",
				kind, name, orig.Name), nil
		}
		to := restoredRefName(orig.Name, newName, name)
		if len(to) > maxObjectNameLength {
			return "", fmt.Sprintf(
				"name %q for the restored copy of %s %q is longer than %d characters",
				to, kind, name, maxObjectNameLength), nil
		}
		renamed[key] = to
		copies = append(copies, ownedRefCopy{kind: kind, from: name, to: to})
		return to, "", nil
	}

	for i := range spec.Env {
		vf := spec.Env[i].ValueFrom
		if vf == nil {
			continue
		}
		if vf.SecretKeyRef != nil {
			to, msg, err := resolve(refKindSecret, vf.SecretKeyRef.Name)
			if err != nil || msg != "" {
				return nil, msg, err
			}
			vf.SecretKeyRef.Name = to
		}
		if vf.ConfigMapKeyRef != nil {
			to, msg, err := resolve(refKindConfigMap, vf.ConfigMapKeyRef.Name)
			if err != nil || msg != "" {
				return nil, msg, err
			}
			vf.ConfigMapKeyRef.Name = to
		}
	}
	if t := spec.Networking.Tunnel; t != nil && t.CredentialsSecretRef != nil && t.CredentialsSecretRef.Name != "" {
		to, msg, err := resolve(refKindSecret, t.CredentialsSecretRef.Name)
		if err != nil || msg != "" {
			return nil, msg, err
		}
		t.CredentialsSecretRef.Name = to
	}
	return copies, "", nil
}

// buildRefCopy reads the original object and returns a new object with the
// same data under the copy's name. Labels, annotations and owner
// references are not carried over.
func (r *RestoreReconciler) buildRefCopy(ctx context.Context, ns string, c ownedRefCopy) (client.Object, error) {
	key := types.NamespacedName{Namespace: ns, Name: c.from}
	meta := metav1.ObjectMeta{Name: c.to, Namespace: ns}
	if c.kind == refKindSecret {
		var src corev1.Secret
		if err := r.Get(ctx, key, &src); err != nil {
			return nil, fmt.Errorf("read Secret %q: %w", c.from, err)
		}
		return &corev1.Secret{ObjectMeta: meta, Type: src.Type, Data: src.Data}, nil
	}
	var src corev1.ConfigMap
	if err := r.Get(ctx, key, &src); err != nil {
		return nil, fmt.Errorf("read ConfigMap %q: %w", c.from, err)
	}
	return &corev1.ConfigMap{ObjectMeta: meta, Data: src.Data, BinaryData: src.BinaryData}, nil
}

// emptyRefObject returns an empty object of the copy's kind, for reads.
func emptyRefObject(kind string) client.Object {
	if kind == refKindSecret {
		return &corev1.Secret{}
	}
	return &corev1.ConfigMap{}
}

// ensureOwnedRefCopies creates each planned copy, controller-owned by the
// restored server. A copy that already exists is accepted only when the
// restored server controls it; any other object under that name fails the
// Restore instead of being adopted or overwritten.
func (r *RestoreReconciler) ensureOwnedRefCopies(
	ctx context.Context, restored *gameplanev1alpha1.GameServer, copies []ownedRefCopy,
) (string, error) {
	for _, c := range copies {
		dst, err := r.buildRefCopy(ctx, restored.Namespace, c)
		if err != nil {
			return "", err
		}
		if err = controllerutil.SetControllerReference(restored, dst, r.Scheme); err != nil {
			return "", fmt.Errorf("set owner on %s %q: %w", c.kind, c.to, err)
		}
		err = r.Create(ctx, dst)
		if err == nil {
			continue
		}
		if !apierrors.IsAlreadyExists(err) {
			return "", fmt.Errorf("create %s %q: %w", c.kind, c.to, err)
		}
		existing := emptyRefObject(c.kind)
		if err = r.Get(ctx, types.NamespacedName{Namespace: restored.Namespace, Name: c.to}, existing); err != nil {
			return "", fmt.Errorf("read %s %q: %w", c.kind, c.to, err)
		}
		if !metav1.IsControlledBy(existing, restored) {
			return fmt.Sprintf("%s %q already exists and is not owned by restored server %q",
				c.kind, c.to, restored.Name), nil
		}
	}
	return "", nil
}

// ensureRestoredRefs makes sure the restored server has its own copies of
// the Secrets and ConfigMaps the original server owns and references. It
// is idempotent, so a Restore reconciled again after creating its server
// finishes any copy an earlier pass didn't. It returns a message when the
// Restore must fail.
func (r *RestoreReconciler) ensureRestoredRefs(
	ctx context.Context, src *gameplanev1alpha1.Backup, restored *gameplanev1alpha1.GameServer,
) (string, error) {
	var orig gameplanev1alpha1.GameServer
	if err := r.Get(ctx, types.NamespacedName{Name: src.Spec.ServerRef.Name, Namespace: restored.Namespace}, &orig); err != nil {
		// Without the original there is nothing left to copy from. The
		// restored server's own reconcile reports any reference it lacks,
		// and the restore deadline ends the wait.
		return "", client.IgnoreNotFound(err)
	}
	scratch := orig.Spec.DeepCopy()
	copies, msg, err := r.planOwnedRefCopies(ctx, &orig, restored.Name, scratch)
	if err != nil || msg != "" {
		return msg, err
	}
	return r.ensureOwnedRefCopies(ctx, restored, copies)
}

package controller

import (
	"context"

	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"sigs.k8s.io/controller-runtime/pkg/client"

	gameplanev1alpha1 "github.com/ValgulNecron/gameplane/operator/api/v1alpha1"

// deleteIfControlledBy removes obj (looked up by its name and namespace)
// only when it exists and gs is its controller. The read is served from
// the informer cache, so a missing object costs no apiserver call, and an
// object someone else created under the same name is left alone. This is
// the same rule reconcileNetworkPolicy applies to its policy.
func (r *GameServerReconciler) deleteIfControlledBy(
	ctx context.Context, gs *gameplanev1alpha1.GameServer, obj client.Object,
) error {
	if err := r.Get(ctx, client.ObjectKeyFromObject(obj), obj); err != nil {
		return client.IgnoreNotFound(err)
	}
	if !metav1.IsControlledBy(obj, gs) {
		return nil
	}
	return client.IgnoreNotFound(r.Delete(ctx, obj))
}
