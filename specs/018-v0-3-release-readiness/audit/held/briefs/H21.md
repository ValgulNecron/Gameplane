package controller

import (

	rbacv1 "k8s.io/api/rbac/v1"

// devRBACDir holds the dev RBAC manifests (`kubectl apply -f
// operator/config/rbac/`), relative to this package.
const devRBACDir = "../../config/rbac"

type devRBACDoc struct {
	Kind     string              `json:"kind"`
	Metadata metav1.ObjectMeta   `json:"metadata"`
	RoleRef  rbacv1.RoleRef      `json:"roleRef"`
	Rules    []rbacv1.PolicyRule `json:"rules"`
}

func loadDevRBAC(t *testing.T) []devRBACDoc {
	t.Helper()
	files, err := filepath.Glob(filepath.Join(devRBACDir, "*.yaml"))
	if err != nil {
		t.Fatalf("list %s: %v", devRBACDir, err)
	}
	if len(files) == 0 {
		t.Fatalf("no manifests found in %s", devRBACDir)
	}
	var docs []devRBACDoc
	for _, f := range files {
		data, err := os.ReadFile(f)
		if err != nil {
			t.Fatalf("read %s: %v", f, err)
		}
		for _, part := range strings.Split("\n"+string(data), "\n---") {
			var d devRBACDoc
			if err := yaml.Unmarshal([]byte(part), &d); err != nil {
				t.Fatalf("parse %s: %v", f, err)
			}
			if d.Kind != "" {
				docs = append(docs, d)
			}
		}
	}
	return docs
}

func ruleGrants(rules []rbacv1.PolicyRule, group, resource, verb string) bool {
	has := func(list []string, want string) bool {
		for _, v := range list {
			if v == want || v == "*" {
				return true
			}
		}
		return false
	}
	for _, r := range rules {
		if has(r.APIGroups, group) && has(r.Resources, resource) && has(r.Verbs, verb) {
			return true
		}
	}
	return false
}

// TestDevRBAC_BindingsReferenceDefinedRoles checks that every binding in
// the dev RBAC manifests points at a ClusterRole or Role that the same
// manifests define.
func TestDevRBAC_BindingsReferenceDefinedRoles(t *testing.T) {
	docs := loadDevRBAC(t)
	clusterRoles := map[string]bool{}
	roles := map[string]bool{} // namespace/name
	for _, d := range docs {
		switch d.Kind {
		case "ClusterRole":
			clusterRoles[d.Metadata.Name] = true
		case "Role":
			roles[d.Metadata.Namespace+"/"+d.Metadata.Name] = true
		}
	}
	bindings := 0
	for _, d := range docs {
		switch d.Kind {
		case "ClusterRoleBinding":
			bindings++
			if d.RoleRef.Kind != "ClusterRole" || !clusterRoles[d.RoleRef.Name] {
				t.Errorf("ClusterRoleBinding %s references %s %q, which the dev manifests don't define",
					d.Metadata.Name, d.RoleRef.Kind, d.RoleRef.Name)
			}
		case "RoleBinding":
			bindings++
			switch d.RoleRef.Kind {
			case "Role":
				if !roles[d.Metadata.Namespace+"/"+d.RoleRef.Name] {
					t.Errorf("RoleBinding %s/%s references Role %q, which the dev manifests don't define in that namespace",
						d.Metadata.Namespace, d.Metadata.Name, d.RoleRef.Name)
				}
			case "ClusterRole":
				if !clusterRoles[d.RoleRef.Name] {
					t.Errorf("RoleBinding %s/%s references ClusterRole %q, which the dev manifests don't define",
						d.Metadata.Namespace, d.Metadata.Name, d.RoleRef.Name)

// TestDevRBAC_ClusterRoleWritesOnlyClusterScopedObjects checks that the
// generated dev ClusterRole grants write verbs only on cluster-scoped
// objects and Events. Namespaced writes belong in the games-namespace
// Role.
func TestDevRBAC_ClusterRoleWritesOnlyClusterScopedObjects(t *testing.T) {
	allowed := map[string]bool{
		"gameplane.local/gametemplates":        true,
		"gameplane.local/gametemplates/status": true,
		"gameplane.local/modules":              true,
		"gameplane.local/modules/status":       true,
		"gameplane.local/modules/finalizers":   true,
		"gameplane.local/modulesources/status": true,
		"gameplane.local/clusters/status":      true,
		"/events":                              true,
	}
	writeVerbs := map[string]bool{
		"create": true, "update": true, "patch": true, "delete": true, "deletecollection": true, "*": true,
	}
	found := false
	for _, d := range loadDevRBAC(t) {
		if d.Kind != "ClusterRole" {
			continue
		}
		found = true
		for _, rule := range d.Rules {
			writes := false
			for _, v := range rule.Verbs {
				if writeVerbs[v] {
					writes = true
				}
			}
			if !writes {
				continue
			}
			for _, g := range rule.APIGroups {
				for _, res := range rule.Resources {
					if !allowed[g+"/"+res] {
						t.Errorf("ClusterRole %s grants %v on %q in group %q cluster-wide; namespaced writes belong in role_namespace.yaml",
							d.Metadata.Name, rule.Verbs, res, g)
					}
				}
			}
		}
	}
	if !found {
		t.Fatal("no ClusterRole in the dev RBAC manifests")
	}
}
