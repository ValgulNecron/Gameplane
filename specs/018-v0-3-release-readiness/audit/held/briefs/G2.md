1. It creates the temp file with `os.CreateTemp`, which makes every written or uploaded file mode `0600`. For an existing regular target it copies the runtime mode from `os.Stat` onto the staged file (`os.Chmod(staged, fi.Mode().Perm())`). A non-literal mode isn't flagged by G302, and `os.Create` passes G307 at its default `0666`. No lint is silenced.

// savePart streams one multipart part into dir under a sanitized name,
// refusing anything larger than limit bytes. It writes to a temp file in
// dir first and renames it over the final name only once the copy
// succeeds, so a failure partway through (a truncated/erroring source, or
// an over-limit part) removes only the temp file — a pre-existing file at
// that name is left untouched instead of being deleted (F-102).
func savePart(dir, filename string, src io.Reader, limit int64) error {
	// Sanitize filename — reject anything that would climb out of dir.
	name := filepath.Base(filename)
	if name == "." || name == ".." || name == string(os.PathSeparator) {
		return errors.New("invalid filename")
	}
	dstPath := filepath.Clean(filepath.Join(dir, name))
	tmp, err := os.CreateTemp(dir, ".upload-*")
	if err != nil {
		return err
	}
	tmpName := tmp.Name()
	// Read one byte past the cap: if that byte materializes the part is
	// over the limit, whatever its multipart headers claimed.
	n, err := io.Copy(tmp, io.LimitReader(src, limit+1))
	closeErr := tmp.Close()
	if err != nil || closeErr != nil {
		_ = os.Remove(tmpName)
		if err != nil {
			return fmt.Errorf("save %q: %w", name, err)
		}
		return fmt.Errorf("save %q: %w", name, closeErr)
	}
	if n > limit {
		_ = os.Remove(tmpName)
		return fmt.Errorf("file %q exceeds %d-byte limit", name, limit)
	}
	if err := os.Rename(tmpName, dstPath); err != nil {
		_ = os.Remove(tmpName)
		return fmt.Errorf("save %q: %w", name, err)
	}
	return nil
}

// savePart streams one multipart part into dir under a sanitized name,
// refusing anything larger than limit bytes. It goes through writeAtomic,
// so a failure partway through (a truncated/erroring source, or an
// over-limit part) leaves nothing behind, and a pre-existing file at that
// name is left untouched instead of being deleted (F-102).
func savePart(dir, filename string, src io.Reader, limit int64) error {
	// Sanitize filename — reject anything that would climb out of dir.
	name := filepath.Base(filename)
	if name == "." || name == ".." || name == string(os.PathSeparator) {
		return errors.New("invalid filename")
	}
	dstPath := filepath.Clean(filepath.Join(dir, name))
	// Read one byte past the cap: if that byte materializes the part is
	// over the limit, whatever its multipart headers claimed.
	err := writeAtomic(dstPath, io.LimitReader(src, limit+1), func(n int64) error {
		if n > limit {
			return fmt.Errorf("file %q exceeds %d-byte limit", name, limit)
		}
		return nil
	})
	if err != nil {
		return fmt.Errorf("save %q: %w", name, err)
	}
	return nil
}

// writeAtomic streams src into dst without ever exposing a partial file
// at dst. The data goes to a file in a private staging directory created
// next to dst, and is renamed over dst only after the copy has finished
// and accept (when non-nil) has approved the byte count. On any failure
// the staging directory is removed and dst is left as it was.
//
// File mode: the staged file is created with os.Create, so a new file
// gets the same mode os.Create has always given it (0o666 minus the
// umask), and the private directory keeps its name from colliding with
// anything. When dst already exists as a regular file, its permission
// bits are copied onto the staged file first, so replacing a file keeps
// its mode. The game container reads this volume under its own uid, so
// the atomic write must not narrow either mode.
func writeAtomic(dst string, src io.Reader, accept func(written int64) error) error {
	dst = filepath.Clean(dst)
	stage, err := os.MkdirTemp(filepath.Dir(dst), ".gameplane-write-")
	if err != nil {
		return err
	}
	defer func() { _ = os.RemoveAll(stage) }()
	staged := filepath.Clean(filepath.Join(stage, "data"))
	f, err := os.Create(staged)
	if err != nil {
		return err
	}
	n, copyErr := io.Copy(f, src)
	closeErr := f.Close()
	if copyErr != nil {
		return copyErr
	}
	if closeErr != nil {
		return closeErr
	}
	if accept != nil {
		if err := accept(n); err != nil {
			return err
		}
	}
	fi, err := os.Stat(dst)
	switch {
	case err == nil && fi.Mode().IsRegular():
		if err := os.Chmod(staged, fi.Mode().Perm()); err != nil {
			return err
		}
	case err != nil && !errors.Is(err, os.ErrNotExist):
		return err
	}
	return os.Rename(staged, dst)
}

// createDefaultMode returns the permission bits os.Create gives a new
// file in this process (0o666 minus the umask), measured on a scratch
// file so the tests don't depend on the umask of the machine running them.
func createDefaultMode(t *testing.T) os.FileMode {
	t.Helper()
	ref := filepath.Join(t.TempDir(), "ref")
	f, err := os.Create(ref)
	if err != nil {
		t.Fatalf("create reference file: %v", err)
	}
	if err := f.Close(); err != nil {
		t.Fatalf("close reference file: %v", err)
	}
	fi, err := os.Stat(ref)
	if err != nil {
		t.Fatalf("stat reference file: %v", err)
	}
	return fi.Mode().Perm()
}
